22: async function apiCall(endpoint, options = {}) {
23:     try {
24:         const response = await fetch(`${API_BASE}${endpoint}`, {
25:             ...options,
26:             headers: {
27:                 'Content-Type': 'application/json',
28:                 ...options.headers,
29:             },
30:         });
31: 
32:         if (!response.ok) {
33:             const error = await response.json();
34:             throw new Error(error.detail || 'API request failed');
35:         }
36: 
37:         return await response.json();
38:     } catch (error) {
39:         console.error('API Error:', error);
40:         showAlert(`Error: ${error.message}`, 'error');
41:         throw error;
42:     }
43: }
