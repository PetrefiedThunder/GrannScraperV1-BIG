228:    job_id: str,
229:    job: ScrapeJob,
230:    concurrent: bool,
231:    incremental: bool
232:) -> ScrapeResult:
233:    """Execute a job (called in background)."""
234:    try:
235:        if incremental:
236:            # Incremental scraping
237:            cache = SmartCache()
238:            incremental_scraper = IncrementalScraper(cache)
239:
240:            # Get URLs to scrape
241:            from scraper.core.engine import ScraperEngine
242:            engine = ScraperEngine()
243:            urls = await engine._generate_urls(job)
244:
245:            # Scrape incrementally
246:            result_data = await incremental_scraper.scrape_incremental(
247:                urls,
248:                lambda url: engine.run_job(job),
249:                ttl_seconds=3600
250:            )
251:
252:            # Convert to ScrapeResult
253:            result = ScrapeResult(
254:                job_id=job_id,
255:                status="success",
256:                items_scraped=len(result_data['new_items']),
257:                pages_visited=result_data['stats']['urls_scraped'],
258:                data=result_data['new_items'],
259:                metadata=result_data['stats']
260:            )
261:
262:        elif concurrent:
263:            # Concurrent scraping
264:            concurrent_scraper = ConcurrentScraper(job, max_workers=10)
265:
266:            # Add URLs
267:            from scraper.core.engine import ScraperEngine
268:            engine = ScraperEngine()
269:            urls = await engine._generate_urls(job)
270:            await concurrent_scraper.add_urls(urls)
271:
272:            # Execute
273:            async def scrape_single(url):
274:                temp_job = ScrapeJob(
275:                    name=job.name,
276:                    start_url=url,
277:                    item_selector=job.item_selector,
278:                    fields=job.fields,
279:                )
280:                r = await engine.run_job(temp_job)
281:                return r.data
282:
283:            result = await concurrent_scraper.run(scrape_single)
284:
285:        else:
286:            # Standard scraping
287:            engine = ScraperEngine()
288:            result = await engine.run_job(job)
289:
290:        # Store result
291:        results_db[job_id] = result
292:
293:        # Export
294:        export_manager = ExportManager(job.export)
295:        await export_manager.export_result(result)

297:        return result
298:
299:    finally:
