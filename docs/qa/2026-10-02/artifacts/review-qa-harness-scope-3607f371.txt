1:// Local-only static server matching rest_server.py's / and /static mounts.
2:const http = require('node:http');
3:const fs = require('node:fs');
4:const path = require('node:path');
5:const root = path.resolve(__dirname, '../../scraper/web/static');
6:const routes = {
7:  '/': ['index.html', 'text/html'],
8:  '/static/index.html': ['index.html', 'text/html'],
9:  '/static/app.js': ['app.js', 'text/javascript'],
10:};
11:http.createServer((req, res) => {
12:  const target = routes[new URL(req.url, 'http://127.0.0.1').pathname];
13:  if (!target) {
14:    res.writeHead(404, { 'Content-Type': 'application/json' });
15:    return res.end(JSON.stringify({ detail: 'Not Found' }));
16:  }
17:  res.writeHead(200, { 'Content-Type': target[1] });
18:  res.end(fs.readFileSync(path.join(root, target[0])));
19:}).listen(18765, '127.0.0.1');
