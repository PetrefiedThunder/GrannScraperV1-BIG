   137	                            })

   146	                    'task': task
   147	                })

   150	            finally:
   151	                self.active_workers -= 1
   152	                self.task_queue.task_done()

   176	    async def _generate_urls(self, job: ScrapeJob) -> list[str]:
   177	        """
   178	        Generate list of URLs to scrape based on pagination config.
   179	
   180	        Args:
   181	            job: ScrapeJob configuration
   182	
   183	        Returns:
   184	            List of URLs
   185	        """
   186	        urls = [job.start_url]
   187	
   188	        pagination = job.pagination
   189	
   190	        if pagination.mode == PaginationMode.NONE:
   191	            return urls
   192	
   193	        elif pagination.mode == PaginationMode.URL_PATTERN:
   194	            # Generate URLs from pattern
   195	            if pagination.url_pattern:
   196	                urls = []
   197	                for page_num in range(
   198	                    pagination.start_page,
   199	                    pagination.start_page + pagination.max_pages
   200	                ):
   201	                    url = pagination.url_pattern.format(page=page_num)
   202	                    urls.append(url)
   203	
   204	        elif pagination.mode == PaginationMode.NEXT_BUTTON:
   205	            # Will handle dynamically during scraping
   206	            # For now, return start URL
   207	            pass
   208	
   209	        elif pagination.mode == PaginationMode.INFINITE_SCROLL:
   210	            # Handled by browser fetcher
   211	            # Return single URL
   212	            pass
   213	
   214	        return urls
