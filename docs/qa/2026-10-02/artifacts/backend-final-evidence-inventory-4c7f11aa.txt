   130	    def _topological_sort_with_levels(self) -> List[List[str]]:
   131	        """
   132	        Topological sort that groups nodes into parallel execution levels.
   133	
   134	        Returns list of lists, where each inner list can execute in parallel.
   135	        """
   136	        # Calculate in-degree
   137	        in_degree = {node_id: 0 for node_id in self.nodes}
   138	        for node in self.nodes.values():
   139	            for dep in node.depends_on:
   140	                in_degree[dep] += 1
   141	
   142	        # Find nodes with no dependencies
   143	        queue = deque([
   144	            node_id for node_id, degree in in_degree.items()
   145	            if degree == 0
   146	        ])
   147	
   148	        levels = []
   149	
   150	        while queue:
   151	            # All nodes in queue can execute in parallel
   152	            current_level = list(queue)
   153	            levels.append(current_level)
   154	
   155	            # Process current level
   156	            next_queue = []
   157	            for node_id in current_level:
   158	                queue.popleft()
   159	
   160	                # Reduce in-degree for dependent nodes
   161	                for other_id, other_node in self.nodes.items():
   162	                    if node_id in other_node.depends_on:
   163	                        in_degree[other_id] -= 1
   164	                        if in_degree[other_id] == 0:
   165	                            next_queue.append(other_id)
   166	
   167	            queue.extend(next_queue)
   168	
   169	        return levels
