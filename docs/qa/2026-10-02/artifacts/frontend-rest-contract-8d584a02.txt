188: @app.post("/api/v1/jobs/{job_id}/run")
189: async def run_job(
190:     job_id: str,
191:     background_tasks: BackgroundTasks,
192:     concurrent: bool = Query(False, description="Use concurrent scraping"),
193:     incremental: bool = Query(False, description="Use incremental scraping")
194: ) -> Dict[str, Any]:
195:     """
196:     Run a scraping job.
197: 
198:     Returns immediately with job status. Use /results endpoint to get results.
199:     """
200:     if job_id not in jobs_db:
201:         raise HTTPException(status_code=404, detail="Job not found")
202: 
203:     # Use lock to prevent race conditions when checking and starting jobs
204:     async with jobs_lock:
205:         if job_id in running_jobs:
206:             raise HTTPException(status_code=400, detail="Job already running")
207: 
208:         job = jobs_db[job_id]
209: 
210:         # Start job in background
211:         task = asyncio.create_task(
212:             _execute_job(job_id, job, concurrent, incremental)
213:         )
214:         running_jobs[job_id] = task
215: 
216:     logger.info(f"Started job: {job_id} (concurrent={concurrent}, incremental={incremental})")
217: 
218:     return {
219:         "job_id": job_id,
220:         "status": "running",
221:         "message": f"Job {job.name} started",
222:         "concurrent": concurrent,
223:         "incremental": incremental
224:     }

226: 
227: async def _execute_job(
228:     job_id: str,
229:     job: ScrapeJob,
230:     concurrent: bool,
231:     incremental: bool
232: ) -> ScrapeResult:
233:     """Execute a job (called in background)."""
234:     try:
235:         if incremental:
236:             # Incremental scraping
237:             cache = SmartCache()
238:             incremental_scraper = IncrementalScraper(cache)
239: 
240:             # Get URLs to scrape
241:             from scraper.core.engine import ScraperEngine
242:             engine = ScraperEngine()
243:             urls = await engine._generate_urls(job)
244: 
245:             # Scrape incrementally
246:             result_data = await incremental_scraper.scrape_incremental(
247:                 urls,
248:                 lambda url: engine.run_job(job),
249:                 ttl_seconds=3600
250:             )
251: 
252:             # Convert to ScrapeResult
253:             result = ScrapeResult(
254:                 job_id=job_id,
255:                 status="success",
256:                 items_scraped=len(result_data['new_items']),
257:                 pages_visited=result_data['stats']['urls_scraped'],
258:                 data=result_data['new_items'],
259:                 metadata=result_data['stats']
260:             )
261: 
262:         elif concurrent:
263:             # Concurrent scraping
264:             concurrent_scraper = ConcurrentScraper(job, max_workers=10)
265: 
266:             # Add URLs
267:             from scraper.core.engine import ScraperEngine
268:             engine = ScraperEngine()
269:             urls = await engine._generate_urls(job)
270:             await concurrent_scraper.add_urls(urls)
271: 
272:             # Execute
273:             async def scrape_single(url):
274:                 temp_job = ScrapeJob(
275:                     name=job.name,
276:                     start_url=url,
277:                     item_selector=job.item_selector,
278:                     fields=job.fields,
279:                 )
280:                 r = await engine.run_job(temp_job)
281:                 return r.data
282: 
283:             result = await concurrent_scraper.run(scrape_single)
284: 
285:         else:
286:             # Standard scraping
287:             engine = ScraperEngine()
288:             result = await engine.run_job(job)
289: 
290:         # Store result
291:         results_db[job_id] = result
292: 
293:         # Export
294:         export_manager = ExportManager(job.export)
295:         await export_manager.export_result(result)
296: 
297:         return result
298: 
299:     finally:
300:         # Remove from running jobs (with lock to prevent race conditions)
301:         async with jobs_lock:
302:             if job_id in running_jobs:
303:                 del running_jobs[job_id]
304:

366: async def analyze_url(request: AnalyzeRequest) -> Dict[str, Any]:
367:     """
368:     Analyze a URL and auto-detect structure.
369: 
370:     Returns detected selectors, fields, and pagination.
371:     """
372:     from scraper.config.models import ScrapeJob
373:     from scraper.core.fetcher_static import StaticFetcher
374:     from scraper.strategies.intelligent_extraction import (
375:         IntelligentExtractor,
376:         SmartPaginationDetector
377:     )
378: 
379:     # Fetch page
380:     temp_job = ScrapeJob(name="analysis", start_url=request.url)
381: 
382:     async with StaticFetcher(temp_job) as fetcher:
383:         soup, html = await fetcher.fetch(request.url)
384: 
385:         if not soup:
386:             raise HTTPException(status_code=400, detail="Failed to fetch URL")
387: 
388:     # Auto-detect
389:     extractor = IntelligentExtractor()
390:     pagination_detector = SmartPaginationDetector()
391: 
392:     item_selector = await extractor.auto_detect_item_container(soup)
393:     detected_fields = await extractor.auto_detect_fields(
394:         soup,
395:         item_selector or "body"
396:     )
397:     pagination_config = await pagination_detector.detect_pagination(soup, request.url)
398: 
399:     return {
400:         "url": request.url,
401:         "item_selector": item_selector,
402:         "fields": detected_fields,
403:         "pagination": pagination_config,
404:         "analysis": {
405:             "html_size": len(html),
406:             "links": len(soup.find_all('a')),
407:             "images": len(soup.find_all('img')),
408:             "title": soup.title.string if soup.title else None
409:         }
